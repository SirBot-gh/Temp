<#
.SYNOPSIS
    Redacts secrets in a Palo Alto Networks PAN-OS configuration / device state XML file.

.DESCRIPTION
    Reads one PAN-OS XML file (running-config.xml, a saved or exported configuration, template/template-config.xml,
    sp/<vsys>/sp-config.xml, or any other XML file from a device state export) and writes a copy in which every
    secret value is replaced by ********. Nothing else changes: element order, attributes, comments, processing
    instructions, whitespace and the XML declaration are preserved, so the output is still a valid PAN-OS config.

    Masked by element / attribute name (case-insensitive, namespace prefixes ignored): phash, password, passwd,
    passphrase, secret, key, psk, authkey, api-key, community, anything ending in -key, -password, -secret,
    -passphrase, -psk, pwd, passwd or keytab, anything containing community, and private-key*.
    Not masked: public-key (certificates / SSH public keys), key-id, ssh-key-id.
    Name/value lists (<entry name="Password"><value>..</value></entry>, as used by ddns-vendor-config, mfa-config
    and HTTP log-forwarding headers/params) mask the <value> when the entry name contains secret, passw, passphrase,
    psk, phash, token, key, community or authorization.
    This covers admin password hashes, IKE pre-shared keys, IPsec manual keys (ESP/AH auth and encryption keys),
    SNMP community strings and SNMPv3 auth/priv passwords, LDAP bind / RADIUS / TACACS+ secrets, NTP authentication
    keys, OSPF/BGP authentication, certificate private keys, Kerberos keytabs, PPPoE and proxy passwords, API keys.

    Only leaf values are masked: a container such as <pre-shared-key> or <auto-key> keeps its child elements and
    the secret leaf inside it (<key>) is masked.

    Files that contain a DOCTYPE are rejected: PAN-OS never emits one, and inline entities could bypass the masking.

    Works in Windows PowerShell 5.1 and PowerShell 7+ (Windows/Linux/macOS), no modules.

.PARAMETER Path
    Input PAN-OS XML file.

.PARAMETER OutFile
    Output file. Default: <input name>.redacted.xml next to the input. The output is never written over the input
    and never replaces an existing file unless -Force is given.

.PARAMETER Force
    Allow overwriting an existing output file, including redacting the input file in place.

.EXAMPLE
    .\Convert-PanDeviceState.ps1 -Path .\running-config.xml
    Writes .\running-config.redacted.xml

.EXAMPLE
    .\Convert-PanDeviceState.ps1 -Path C:\exports\FW-01\template\template-config.xml -OutFile C:\share\template-config.xml

.EXAMPLE
    Get-ChildItem C:\exports -Recurse -Filter *.xml | ForEach-Object { .\Convert-PanDeviceState.ps1 -Path $_.FullName }
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory = $true, Position = 0)]
    [string]$Path,
    [Parameter(Position = 1)]
    [string]$OutFile,
    [switch]$Force
)

Set-StrictMode -Version 1.0
$ErrorActionPreference = 'Stop'

$script:SecretPattern = '^(phash|password|passwd|passphrase|secret|key|psk|authkey|api-key|community|.*-key|.*-password|.*-secret|.*-passphrase|.*-psk|.*pwd|.*passwd|.*community.*|private-key.*|.*keytab)$'
$script:SecretExclude = '^(public-key|key-id|ssh-key-id)$'
# Entry names in name/value lists (ddns-vendor-config, mfa-config, http headers/params) are vendor labels such as
# "Password", "Duo Secret Key", "Token", "Authorization"; only the <value> leaf of such an entry is masked.
$script:SecretEntryNamePattern = '(secret|passw|passphrase|psk|phash|token|key|community|authorization)'
$script:Mask = '********'
$script:ScriptVersion = '2.0'

function Protect-Secrets {
    param([System.Xml.XmlDocument]$Doc)
    $count = 0
    $all = $Doc.SelectNodes('//*')
    foreach ($el in @($all)) {
        if ($el.LocalName -match $script:SecretPattern -and $el.LocalName -notmatch $script:SecretExclude) {
            # Leaf values only; containers (e.g. pre-shared-key > key) are handled via their matching children.
            $hasElementKids = $false
            foreach ($ch in $el.ChildNodes) { if ($ch.NodeType -eq [System.Xml.XmlNodeType]::Element) { $hasElementKids = $true; break } }
            if (-not $hasElementKids) {
                foreach ($t in @($el.ChildNodes)) {
                    if (($t.NodeType -eq [System.Xml.XmlNodeType]::Text -or $t.NodeType -eq [System.Xml.XmlNodeType]::CDATA) -and $t.Value.Trim() -ne '') { $t.Value = $script:Mask; $count++ }
                }
            }
            # xml:space / xml:lang / xmlns declarations are never secrets and masking them makes the output invalid XML.
            foreach ($a in @($el.Attributes)) { if ($a.Name -ne 'name' -and $a.Prefix -ne 'xml' -and $a.Prefix -ne 'xmlns' -and $a.Name -ne 'xmlns') { $a.Value = $script:Mask } }
        }
        # Name/value lists (ddns-vendor-config, mfa-config, http headers/params): <entry name="Password"><value>..</value></entry>
        if ($el.LocalName -eq 'entry' -and $el.HasAttribute('name') -and $el.GetAttribute('name') -match $script:SecretEntryNamePattern) {
            foreach ($v in @($el.ChildNodes)) {
                if ($v.NodeType -ne [System.Xml.XmlNodeType]::Element -or $v.LocalName -ne 'value') { continue }
                foreach ($t in @($v.ChildNodes)) {
                    if (($t.NodeType -eq [System.Xml.XmlNodeType]::Text -or $t.NodeType -eq [System.Xml.XmlNodeType]::CDATA) -and $t.Value.Trim() -ne '') { $t.Value = $script:Mask; $count++ }
                }
            }
        }
        foreach ($a in @($el.Attributes)) {
            if ($a.LocalName -match $script:SecretPattern -and $a.LocalName -notmatch $script:SecretExclude) { $a.Value = $script:Mask; $count++ }
        }
    }
    return $count
}

# ---------------------------------------------------------------- main
$inputItem = Get-Item -LiteralPath $Path
if ($inputItem.PSIsContainer) { throw "Path must be a single XML file, not a folder: $Path" }
$inputFull = $inputItem.FullName

if (-not $OutFile) {
    $OutFile = Join-Path $inputItem.DirectoryName ([System.IO.Path]::GetFileNameWithoutExtension($inputItem.Name) + '.redacted' + $inputItem.Extension)
}
$OutFile = [System.IO.Path]::GetFullPath($(if ([System.IO.Path]::IsPathRooted($OutFile)) { $OutFile } else { Join-Path (Get-Location).ProviderPath $OutFile }))
$inPlace = [string]::Equals($OutFile, $inputFull, [System.StringComparison]::OrdinalIgnoreCase)
if ($inPlace -and -not $Force) { throw "OutFile is the input file; use -Force to redact it in place: $OutFile" }
if (-not $inPlace -and (Test-Path -LiteralPath $OutFile) -and -not $Force) { throw "OutFile already exists; use -Force to overwrite it: $OutFile" }

# Remember whether the input starts with a UTF-8 BOM and whether it uses CRLF, so the output can match it
# (the XML parser normalises CRLF to LF while reading).
$hasBom = $false
$fs = [System.IO.File]::OpenRead($inputFull)
try { $b = New-Object byte[] 3; $n = $fs.Read($b, 0, 3); $hasBom = ($n -eq 3 -and $b[0] -eq 0xEF -and $b[1] -eq 0xBB -and $b[2] -eq 0xBF) } finally { $fs.Dispose() }
$hasCrlf = ([System.IO.File]::ReadAllText($inputFull)).Contains("`r`n")

$doc = New-Object System.Xml.XmlDocument
$doc.PreserveWhitespace = $true
$doc.XmlResolver = $null
# XmlDocument.Load(path) parses an inline DTD and keeps general entities as EntityReference nodes whose
# text Protect-Secrets cannot rewrite (e.g. <phash>&x;</phash> leaks). PAN-OS never emits a DOCTYPE, so refuse it.
$xrs = New-Object System.Xml.XmlReaderSettings
$xrs.DtdProcessing = [System.Xml.DtdProcessing]::Prohibit
try {
    $xr = [System.Xml.XmlReader]::Create($inputFull, $xrs)
    try { $doc.Load($xr) } finally { $xr.Close() }
} catch { throw ("Cannot parse '" + $inputFull + "' as XML: " + $_.Exception.Message) }

$masked = Protect-Secrets $doc

# Write it back as it was read: the original XML declaration verbatim (or none), same encoding/BOM, original
# whitespace and newlines. (.NET still normalises empty tags to "<x />" and attribute quotes to double quotes.)
$decl = $null
foreach ($c in $doc.ChildNodes) { if ($c.NodeType -eq [System.Xml.XmlNodeType]::XmlDeclaration) { $decl = $c } }
$enc = New-Object System.Text.UTF8Encoding($false)
if ($decl -and $decl.Encoding -and $decl.Encoding -notmatch '^utf-?8$') { $enc = [System.Text.Encoding]::GetEncoding($decl.Encoding) }
$xws = New-Object System.Xml.XmlWriterSettings
$xws.Encoding = $enc
$xws.Indent = $false
$xws.NewLineHandling = [System.Xml.NewLineHandling]::None
if ($hasCrlf) { $xws.NewLineHandling = [System.Xml.NewLineHandling]::Replace; $xws.NewLineChars = "`r`n" }
$xws.OmitXmlDeclaration = $true
$ms = New-Object System.IO.MemoryStream
$xw = [System.Xml.XmlWriter]::Create($ms, $xws)
try { $doc.Save($xw) } finally { $xw.Close() }
$outDir = [System.IO.Path]::GetDirectoryName($OutFile)
if (-not (Test-Path -LiteralPath $outDir)) { [void][System.IO.Directory]::CreateDirectory($outDir) }
$out = [System.IO.File]::Create($OutFile)
try {
    if ($hasBom) { $pre = $enc.GetPreamble(); if ($pre.Length -eq 0) { $pre = [byte[]](0xEF, 0xBB, 0xBF) }; $out.Write($pre, 0, $pre.Length) }
    if ($decl) { $d = $enc.GetBytes('<?xml ' + $decl.InnerText + '?>'); $out.Write($d, 0, $d.Length) }
    $body = $ms.ToArray(); $out.Write($body, 0, $body.Length)
} finally { $out.Dispose(); $ms.Dispose() }

Write-Host ('Masked ' + $masked + ' value(s); wrote ' + $OutFile)
